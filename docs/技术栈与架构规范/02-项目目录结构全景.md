# 二、项目目录结构全景

```text
ecommerce-competitor-analysis/
├── apps/
│   ├── backend/                     # [服务端] NestJS API 与 BullMQ 任务进程
│   │   ├── prisma/                  # MySQL 数据库模型与迁移
│   │   │   ├── schema.prisma        # 35+ 张核心业务表结构
│   │   │   └── seed.ts              # 幂等初始数据（租户、超管、权限字典）
│   │   ├── src/                     # NestJS 核心业务源码
│   │   │   ├── admin/               # 租户与系统设置后台
│   │   │   ├── agent/               # AI 智能体编排与工作流
│   │   │   ├── auth/                # JWT 鉴权与守卫（Permission/DataScope）
│   │   │   ├── jobs/                # 任务定义与模板
│   │   │   ├── queue/               # BullMQ 队列与消费者
│   │   │   ├── reports/             # 竞品报告、价格带聚类、A+ 图文分析
│   │   │   ├── main.ts              # HTTP API 服务入口 (:8787)
│   │   │   └── worker.ts            # BullMQ Worker 独立进程入口
│   │   └── package.json
│   │
│   ├── desktop/                     # [桌面端] Electron 桌面客户端
│   │   ├── electron.vite.config.ts  # Electron-Vite 构建配置
│   │   ├── prisma/                  # 桌面端本地 SQLite 模型
│   │   │   └── schema.prisma        # 本地离线采集批次记录
│   │   ├── src/
│   │   │   ├── main/                # Electron 主进程源码
│   │   │   │   ├── index.ts         # 主窗口生命周期管理
│   │   │   │   ├── collection-handlers.ts # 采集任务 IPC 处理器
│   │   │   │   ├── python-runner.ts # Python 运行时与 CDP 启动器
│   │   │   │   └── sqlite.ts        # 本地 SQLite 客户端
│   │   │   ├── preload/             # 预加载安全网桥
│   │   │   │   ├── index.ts         # 暴露 window.desktop 安全 API
│   │   │   │   └── index.d.ts
│   │   │   └── renderer/            # 业务 UI 渲染层 (全量前端页面)
│   │   │       ├── src/
│   │   │       │   ├── api/         # Axios 客户端与全局拦截器
│   │   │       │   ├── components/  # 公共业务组件（选择器、报告卡片等）
│   │   │       │   ├── pages/       # 各业务页面（竞品报告、复刻、采集等）
│   │   │       │   ├── utils/       # 进度解析、数值格式化等工具函数
│   │   │       │   ├── env.d.ts     # window.desktop 全局 TypeScript 声明
│   │   │       │   ├── main.tsx     # React 根入口，引入 Arco 样式与路由
│   │   │       │   └── styles.css   # 全局基础样式与少量定制 CSS
│   │   │       └── index.html
│   │   └── package.json
│   │
│   └── legacy/                      # 旧版单体代码（仅作历史清理/参考）
│
├── docs/                            # 项目文档与技术规约
│   ├── 技术栈与架构规范/            # [本目录] 系统架构、技术栈与 AI 索引
│   │   ├── index.yaml               # 模块索引清单
│   │   └── *.md                     # 独立模块技术手册
│   └── 开发任务/                    # 历史需求、任务拆解与交付记录
│
├── skills/                          # 店透视 RPA 与自动化采集 Python 脚本
├── scripts/                         # 环境自检 (check-env) 与运维脚本
├── package.json                     # Monorepo 根配置
└── pnpm-workspace.yaml              # pnpm 聚合工作区定义
```
